\begin{frame}{Мотивация}

\begin{block}{Цель}
Разработка метода и архитектуры нейронной сети для быстрой двунаправленной трансляции между нативными и контрастными КТ сериями.
\end{block}

\begin{alertblock}{Проблема}
Диффузионные модели требуют сотни шагов сэмплирования.
Существующие методы допускают трансляцию только в одну сторону.
\end{alertblock}

\begin{exampleblock}{Предлагаемое решение}
Фреймворк \textbf{Medical Flow Matching (MFM)} с архитектурой \textbf{TimeResNet}:
инъекция временного параметра в каждый блок и свёрточный self-attention в узком горлышке.
\end{exampleblock}

\end{frame}

\begin{frame}{Определения и Conditional Flow Matching}
\small

\begin{defn}[Пространство КТ срезов и поток]
$\mathcal{X} = [-1,1]^{H \times W}$ --- нормализованные КТ срезы; $\pi_N, \pi_C$ --- распределения нативных и контрастных серий; $\mathcal{D} = \{(\mathbf{x}_N^{(i)}, \mathbf{x}_C^{(i)})\}_{i=1}^{n}$.
Векторное поле $f_\theta : \mathcal{X} \times [0,1] \to \mathbb{R}^{H \times W}$ задаёт поток $\varphi_t$:
$d\mathbf{x}_t/dt = f_\theta(\mathbf{x}_t, t)$, $\varphi_{1\#}\pi_N = \pi_C$.
\end{defn}

\vspace{-0.1cm}
\begin{block}{Линейный интерполянт и функция потерь}
Для $(\mathbf{x}_0, \mathbf{x}_1) \sim \pi_N \times \pi_C$:
$\mathbf{x}_t = (1 - t)\mathbf{x}_0 + t\mathbf{x}_1$, $\mathbf{v}^* = \mathbf{x}_1 - \mathbf{x}_0$.
\[
  \mathcal{L}_{\mathrm{FM}}(\theta) = \mathbb{E}_{t \sim \mathcal{U}[0,1]} \mathbb{E}_{(\mathbf{x}_0,\mathbf{x}_1) \sim \pi_N \times \pi_C} \bigl[\lVert f_\theta(\mathbf{x}_t, t) - \mathbf{v}^* \rVert_2^2\bigr], \quad \theta^* = \arg\min_\theta \mathcal{L}_{\mathrm{FM}}(\theta).
\]
\end{block}

\vspace{-0.1cm}
\begin{prop}[Двунаправленность]
Для $f_{\theta^*}$ и любого $\mathbf{x} \in \mathcal{X}$:
$\hat{\mathbf{x}}_C = \mathbf{x} + \int_0^1 f_{\theta^*}(\mathbf{x}_t, t)\,dt$, $\hat{\mathbf{x}}_N = \mathbf{x} + \int_1^0 f_{\theta^*}(\mathbf{x}_t, t)\,dt$ --- одна сеть реализует оба отображения $\pi_N \leftrightarrow \pi_C$.
\end{prop}

\end{frame}

\begin{frame}{Per-block time injection: формализация}

\begin{columns}[c]
\begin{column}{0.46\textwidth}
\centering
\includegraphics[trim={0 422 0 30},clip,height=0.78\textheight,keepaspectratio]{images/Embeding.pdf}
\end{column}
\begin{column}{0.54\textwidth}

\begin{defn}[Time injection]
Оператор $\mathcal{M}_l : \mathbb{R}^{C_l \times H_l \times W_l} \to \mathbb{R}^{C_l \times H_l \times W_l}$
в блоке $l = 1, \ldots, L$:
\[
  \mathcal{M}_l(\mathbf{h}; t) = \mathbf{A}_l(t) \odot \mathbf{h} + \mathbf{B}_l(t),
\]
\[
  (\mathbf{A}_l, \mathbf{B}_l) = W_l \cdot \mathrm{MLP}\bigl(\gamma(t)\bigr) \in \mathbb{R}^{2C_l},
\]
где $\gamma : [0,1] \to \mathbb{R}^{d_{\mathrm{emb}}}$ --- синусоидальное кодирование, $W_l \in \mathbb{R}^{2C_l \times d_{\mathrm{mlp}}}$ --- обучаемая проекция блока~$l$.
\end{defn}

\end{column}
\end{columns}

\end{frame}

\begin{frame}{Self-Attention в узком горлышке: формализация}

\begin{columns}[c]
\begin{column}{0.52\textwidth}
\centering
\includegraphics[width=1.0\linewidth]{images/self_attention.pdf}
\end{column}
\begin{column}{0.48\textwidth}

\begin{defn}[Self-attention]
Пусть $\mathbf{Z} \in \mathbb{R}^{C \times H' \times W'}$ --- вход.
\[
  Q = W_Q * \mathbf{Z},\;\; K = W_K * \mathbf{Z},\;\; V = W_V * \mathbf{Z},
\]
$W_Q, W_K, W_V$ --- свёртки $1{\times}1$, $N = H' \!\cdot\! W'$.
\[
  \mathrm{Attn}(\mathbf{Z}) = \mathbf{Z} + V \cdot \mathrm{softmax}\!\Bigl(\frac{Q^\top K}{\sqrt{C}}\Bigr).
\]
Сложность: $\mathcal{O}(N^2 C)$,\; $N = \frac{HW}{4^k} \ll HW$.
\end{defn}

\end{column}
\end{columns}

\end{frame}

\begin{frame}{Свойства архитектуры TimeResNet}
\small

\begin{prop}[Сохранение временного сигнала]
При per-block injection временной сигнал имеет прямой ненулевой канал на каждой глубине:
\[
  \frac{\partial \mathcal{L}}{\partial t} = \sum_{l=1}^{L} \frac{\partial \mathcal{L}}{\partial \mathbf{h}_l} \cdot \frac{\partial \mathcal{M}_l}{\partial t} \neq 0, \qquad \frac{\partial \mathcal{M}_l}{\partial t} \neq 0 \;\; \forall\, l.
\]
\end{prop}

\begin{prop}[Вычислительная сложность self-attention]
Пусть $k$ --- глубина encoder, $N_{\mathrm{bottle}} = HW / 4^k$. Тогда:
\[
  \frac{\mathcal{O}(N_{\mathrm{bottle}}^2\, C)}{\mathcal{O}((HW)^2\, C)} = \frac{1}{4^{2k}}.
\]
При $k = 4$: сложность в $\approx 65\,000$ раз меньше full-resolution attention.
\end{prop}

\end{frame}

\begin{frame}{Результаты: N $\to$ C и MAE по органам (held-out)}

\begin{prop}[Двунаправленная трансляция без переобучения]
Пусть $f_{\theta^*}$ обучена на $\mathcal{L}_{\mathrm{FM}}$ с симметричной выборкой пар. Тогда
$f_{\theta^*} = \arg\min_f \bigl[\mathcal{L}_{\mathrm{FM}}^{N \to C}(f) + \mathcal{L}_{\mathrm{FM}}^{C \to N}(f)\bigr]$~--- переобучение для обратного направления не требуется. На аорте при синтезе контраста MFM в \textbf{15$\times$} точнее SMILE.
\end{prop}

\begin{frame}{Выносится на защиту}
\small

\begin{enumerate}\setlength{\itemsep}{0.25em}
  \item Фреймворк \textbf{Medical Flow Matching}: $\theta^* = \arg\min_\theta \mathcal{L}_{\mathrm{FM}}(\theta)$ с линейным интерполянтом~--- математическая формализация задачи переноса меры $\pi_N \leftrightarrow \pi_C$.

  \item \textbf{Утверждение о двунаправленности}: $f_{\theta^*} = \arg\min_f \bigl[\mathcal{L}_{\mathrm{FM}}^{N\to C}(f) + \mathcal{L}_{\mathrm{FM}}^{C\to N}(f)\bigr]$~--- единственная сеть реализует оба направления без переобучения.

  \item Архитектура \textbf{TimeResNet} с теоретически обоснованными компонентами:
  \begin{itemize}\setlength{\itemsep}{0pt}
      \item \textbf{Сохранение временного сигнала}: при $\mathcal{M}_l(\mathbf{h}; t) = \mathbf{A}_l(t)\odot\mathbf{h} + \mathbf{B}_l(t)$ выполнено $\partial\mathcal{L}/\partial t \neq 0$ на всех глубинах $l = 1, \ldots, L$;
      \item \textbf{Сложность боттлнекового self-attention}: $\mathcal{O}(N_{\mathrm{bottle}}^2 C)$, в $4^{2k}$ раз дешевле полноразмерного.
  \end{itemize}
\end{enumerate}
