\begin{frame}{Проблематика}
\begin{itemize}

    \item \textbf{Цель:} решается проблема трансляции модальности между сериями КТ изображений одного пациента.
    
    \item \textbf{Исследуемая проблема:} Диффузионные модель для КТ изображений требует огромные вычислительные ресурсы. Возможность направлять процесс трансляции только в одну сторону.
    
    \item \textbf{Метод решения:} Аппроксимация векторного поля между двумя распределениями $\pi_{0}$ и $\pi_{1}$. Предложен метод инъекции временных в эмбедингов в блок нейронной сети.
\end{itemize}
\end{frame}
% ------------------------------------------------------------------------


\begin{frame}{Flow Matching}

\begin{block}{Постановка задачи}
Даны распределения $\pi_0$ (источник) и $\pi_1$ (цель) в $\mathbb R^d$.
Найти векторное поле $v_\theta:\mathbb R^{512 \times 512}\times[0,1]\!\to\!\mathbb R^{512 \times 512}$,  
такое что решение ОДУ
\[
\frac{d X_t}{dt}=v_\theta(X_t,t),\qquad X_0\sim \pi_0,
\]
переносит $\pi_0$ в $\pi_1$ при $t=1$.
\end{block}

\begin{block}{Путь}
Линейная траектория
\[
X_t=(1-t)X_0+tX_1,\quad (X_0,X_1)\sim \pi_0\times \pi_1.
\]
Тогда таргетное поле постоянно:
\[
v^{\star}(x,t)=X_1-X_0\;\;(\text{не зависит от }t).
\]
\end{block}

\end{frame}



% ------------------------------------------------------------------------
\begin{frame}{Flow Matching}
\begin{block}{Функция потерь}
    Обучаем $v_{\theta}$ аппроксимировать $v_{target}$ 
в среднем по промежуточным скоростям:
\begin{equation*}
    \mathcal{L}(\theta) = \text{  } \mathbb{E}_{t \sim \mathcal{U}[0,1]}
  \mathbb{E}_{X_0, X_1 \sim \pi_0 \times \pi_1} \bigl[\|v_{\theta}(X_t, t) - v^*\|^{2}\bigr] 
\end{equation*}
\begin{equation*}
    \theta^* = \arg\min_{\theta} \mathcal{L}(\theta)
\end{equation*}
\end{block}

\begin{block}{Трансляция}
Объект из $\pi_0$ и численно интегрируем  $\dot X_t = v_{\theta}(X_t, t)$ от $t = 0$ до $t = 1$.

Объект из $\pi_1$ и численно интегрируем $\dot X_t = v_{\theta}(X_t, t)$ от $t = 1$ до $t = 0$.

\end{block}
\end{frame}

\begin{frame}{Постановка экперимента}

\begin{enumerate}
    \item Дана выборка $k$ объектов
    \[
        \mathcal{D} = \{x_{0i}, x_{1i}\},\ x_{0i}, x_{1i}  \in \pi_0 \times \pi_1 \ , i = 1, \dots, k,\
  \]
    где $x_i, y_i$ -- аксиальных срезы Нативного и Контрастного изображения пациента. 120 исследований (52561 изображений), 80 (34992) - обучение, 20 (8197) - тестовая, 20 (9372) - валидационная. 

    \item Дополнительно проводится регистрация изображений с контрастом на нативное для уменьшения шума, связанного с движениями пациента, проводится локализация объекта исследования с помощью сегментационной модели для устранения артефактов.
  

    \item Решается задача оптимизации
    \begin{equation*}
        \mathbf{\theta}^* = \arg\min_{\mathbf{\theta} \in \mathbf{\Theta}} \mathcal{L}(X, \mathbf{\theta}),
    \end{equation*}
    где $\Theta$ -- пространство нейросетей и предложенной архитектуры: TimeResNet, DiffusionNet, SwinUNETR, UNet, SegResNet.
\end{enumerate}
\end{frame}
